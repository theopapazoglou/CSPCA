def rbf_kernel(Y, sigma):
    dist_matrix = cdist(Y, Y, 'euclidean')
    K = np.exp(-dist_matrix**2 / (2 * sigma**2))
    return K

def spca_hsic(X, Y, K, q):
    C = X.T @ K @ X
    eigenvalues, eigenvectors = linalg.eigh(C)
    sorted_indices = np.argsort(eigenvalues)[::-1]
    eigenvalues = eigenvalues[sorted_indices]
    eigenvectors = eigenvectors[:, sorted_indices]
    W = eigenvectors[:, :q]
    return {'W': W, 'eigenvalues': eigenvalues[:q]}

def cspca(X, Y, q, lambda_):
    XtX = X.T @ X
    XtY = X.T @ Y
    YtX = Y.T @ X
    C = XtY @ YtX + lambda_ * XtX
    eigenvalues, eigenvectors = linalg.eigh(C)
    sorted_indices = np.argsort(eigenvalues)[::-1]
    eigenvalues = eigenvalues[sorted_indices]
    eigenvectors = eigenvectors[:, sorted_indices]
    W = eigenvectors[:, :q]
    return {'W': W, 'eigenvalues': eigenvalues[:q]}

def run_all_methods_analysis(data, n_components_list=[2,4,6,8,10],
                           lambda_list=[0.001, 0.01, 0.1],  
                           kappa_list=[0.001, 0.01, 0.1], 
                           threshold_list=[0.05, 0.1, 0.15], 
                           seed=1924):
    Y = data[:, 0:1]  
    X = data[:, 1:]
    p = X.shape[1]

    results = pd.DataFrame({
        'n_components': n_components_list,
        'variance_explained_pcr': onp.zeros(len(n_components_list)),
        'mse_PCR': onp.zeros(len(n_components_list)),
        'covariance_explained_PCR': onp.zeros(len(n_components_list)),
        'variance_explained_pls': onp.zeros(len(n_components_list)),
        'mse_PLS': onp.zeros(len(n_components_list)),
        'variance_explained_hsic': onp.zeros(len(n_components_list)),
        'mse_Hsic': onp.zeros(len(n_components_list)),
        'covariance_explained_Hsic': onp.zeros(len(n_components_list)),
        'variance_explained_bair': onp.zeros(len(n_components_list)),
        'mse_Bair': onp.zeros(len(n_components_list)),
        'covariance_explained_Bair': onp.zeros(len(n_components_list)),
        'variance_explained_cspca': onp.zeros(len(n_components_list)),
        'mse_CSPCA': onp.zeros(len(n_components_list)),
        'covariance_explained_CSPCA': onp.zeros(len(n_components_list)),
        'variance_explained_lspca': onp.zeros(len(n_components_list)),
        'mse_LSPCA': onp.zeros(len(n_components_list)),
        'covariance_explained_LSPCA': onp.zeros(len(n_components_list)),
        'best_threshold': onp.zeros(len(n_components_list)),  
        'best_kappa': onp.zeros(len(n_components_list)),
        'best_lambda': onp.zeros(len(n_components_list))
    })

    for i, n_components in enumerate(n_components_list):
        metrics = {key: [] for key in results.columns if key not in ['n_components', 'best_threshold', 'best_kappa', 'best_lambda']}
        best_params = {
            'threshold': {k: [] for k in ['value', 'mse']},
            'kappa': {k: [] for k in ['value', 'mse']},
            'lambda': {k: [] for k in ['value', 'mse']}
        }

        n = X.shape[0]
        X_train, X_test, Y_train, Y_test = train_test_split(X, Y, test_size=0.2, random_state=seed)
        X_train, X_val, Y_train, Y_val = train_test_split(X_train, Y_train, test_size=0.25, random_state=seed) 
         
        scaler_X = StandardScaler()
        X_train = scaler_X.fit_transform(X_train)
        X_val = scaler_X.transform(X_val)
        X_test = scaler_X.transform(X_test)
        scaler_Y = StandardScaler()
        Y_train = scaler_Y.fit_transform(Y_train)
        Y_val = scaler_Y.transform(Y_val)
        Y_test = scaler_Y.transform(Y_test)

        X_train_np = np.array(X_train)
        X_val_np = np.array(X_val)
        Y_train_np = np.array(Y_train)
        Y_val_np = np.array(Y_val)
        

        # PCR    
        pca = PCA(n_components=n_components)
        X_train_pca = pca.fit_transform(X_train)
        X_test_pca = pca.transform(X_test)
        var_explained = onp.sum(pca.explained_variance_ratio_)
        metrics['variance_explained_pcr'].append(var_explained)
        lr = LinearRegression()
        lr.fit(X_train_pca, Y_train)
        Y_pred = lr.predict(X_test_pca)
        metrics['mse_PCR'].append(mean_squared_error(Y_test, Y_pred))
        cov_xy = onp.cov(X_train.T, Y_train.T)[:-1, -1]
        cov_zy = onp.cov(X_train_pca.T, Y_train.T)[:-1, -1]
        metrics['covariance_explained_PCR'].append(onp.sum(cov_zy**2) / onp.sum(cov_xy**2))

        # PLS
        pls = PLSRegression(n_components=n_components)
        pls.fit(X_train, Y_train)
        Y_pred_pls = pls.predict(X_test)
        var_expl_pls = onp.var(pls.x_scores_, axis=0).sum() / onp.var(X_train, axis=0).sum()
        metrics['variance_explained_pls'].append(var_expl_pls)
        metrics['mse_PLS'].append(mean_squared_error(Y_test, Y_pred_pls))

        # SPCA_HSIC
        sigma = 0.1  
        K = rbf_kernel(Y_train, sigma)
        spca_result = spca_hsic(X_train, Y_train, K, n_components)
        W_hsic = spca_result['W']
        Z_train_bar = X_train @ W_hsic
        Z_test_bar = X_test @ W_hsic
        total_var = onp.trace(X_train.T @ X_train)
        expl_var = onp.trace(W_hsic.T @ X_train.T @ X_train @ W_hsic)
        metrics['variance_explained_barshan'].append(expl_var / total_var)
        lr_barshan = LinearRegression().fit(Z_train_bar, Y_train)
        Y_pred_bar = lr_barshan.predict(Z_test_bar)
        metrics['mse_Barshan'].append(mean_squared_error(Y_test, Y_pred_bar))
        cov_zy = onp.cov(Z_train_bar.T, Y_train.T)[:-1, -1]
        metrics['covariance_explained_Barshan'].append(onp.sum(cov_zy**2) / onp.sum(cov_xy**2))

        # Bair's Method
        best_threshold_mse = float('inf')
        best_threshold = None
        for threshold in threshold_list:
                feature_scores = onp.abs(onp.corrcoef(X_train.T, Y_train.T)[:-1, -1])
                selected_features = feature_scores >= threshold
                if onp.sum(selected_features) == 0:
                    continue  # Skip if no features selected
                
                X_train_selected = X_train[:, selected_features]
                X_val_selected = X_val[:, selected_features]
                pca_bair = PCA(n_components=min(n_components, X_train_selected.shape[1]))
                X_train_bair = pca_bair.fit_transform(X_train_selected)
                X_val_bair = pca_bair.transform(X_val_selected)
                lr_bair = LinearRegression().fit(X_train_bair, Y_train)
                Y_pred_bair = lr_bair.predict(X_val_bair)
                current_mse = mean_squared_error(Y_val, Y_pred_bair)
                
                if current_mse < best_threshold_mse:
                    best_threshold_mse = current_mse
                    best_threshold = threshold
            
        if best_threshold is not None:
                feature_scores = onp.abs(onp.corrcoef(X_train.T, Y_train.T)[:-1, -1])
                selected_features = feature_scores >= best_threshold
                X_train_selected = X_train[:, selected_features]
                X_test_selected = X_test[:, selected_features]
                pca_bair = PCA(n_components=min(n_components, X_train_selected.shape[1]))
                X_train_bair = pca_bair.fit_transform(X_train_selected)
                X_test_bair = pca_bair.transform(X_test_selected)
                expl_var_bair = onp.trace(X_train_bair.T @ X_train_bair)
                metrics['variance_explained_bair'].append(expl_var_bair / total_var)
                lr_bair = LinearRegression().fit(X_train_bair, Y_train)
                Y_pred_bair = lr_bair.predict(X_test_bair)
                metrics['mse_Bair'].append(mean_squared_error(Y_test, Y_pred_bair))
                cov_zy = onp.cov(X_train_bair.T, Y_train.T)[:-1, -1]
                metrics['covariance_explained_Bair'].append(onp.sum(cov_zy**2) / onp.sum(cov_xy**2))
                best_params['threshold']['value'].append(best_threshold)
                best_params['threshold']['mse'].append(best_threshold_mse)
        else:
                metrics['variance_explained_bair'].append(0)
                metrics['mse_Bair'].append(float('inf'))
                metrics['covariance_explained_Bair'].append(0)
                best_params['threshold']['value'].append(threshold_list[0])
                best_params['threshold']['mse'].append(float('inf'))

        best_kappa_mse = float('inf')
        best_kappa = None
        for kappa in kappa_list:
                spca_sup = supervised_pca(X_train, Y_train, n_components, kappa)
                W_sup = spca_sup['W']
                Z_train_sup = X_train @ W_sup
                Z_val_sup = X_val @ W_sup
                lr_sup = LinearRegression().fit(Z_train_sup, Y_train)
                Y_pred_sup = lr_sup.predict(Z_val_sup)
                current_mse = mean_squared_error(Y_val, Y_pred_sup)
                
                if current_mse < best_kappa_mse:
                    best_kappa_mse = current_mse
                    best_kappa = kappa

        # CSPCA
        spca_sup = cspca(X_train, Y_train, n_components, best_kappa)
        W_cspca = spca_sup['W']
        Z_train_sup = X_train @ W_cspca
        Z_test_sup = X_test @ W_cspca
        expl_var_sup = onp.trace(W_cspca.T @ X_train.T @ X_train @ W_cspca)
        metrics['variance_explained_supervised'].append(expl_var_sup / total_var)
        lr_sup = LinearRegression().fit(Z_train_sup, Y_train)
        Y_pred_sup = lr_sup.predict(Z_test_sup)
        metrics['mse_Supervised'].append(mean_squared_error(Y_test, Y_pred_sup))
        cov_zy = onp.cov(Z_train_sup.T, Y_train.T)[:-1, -1]
        metrics['covariance_explained_Supervised'].append(onp.sum(cov_zy**2) / onp.sum(cov_xy**2))
        best_params['kappa']['value'].append(best_kappa)
        best_params['kappa']['mse'].append(best_kappa_mse)

        # LSPCA
        best_lambda_mse = float('inf')
        best_lambda = None
        for lambda in lambda_list:
                manifold = Product([Grassmann(p, n_components), Euclidean(n_components, 1)])
                @pymanopt.function.autograd(manifold)
                def cost(L, w):
                    XL = X_train_np @ L
                    lr_resid = Y_train_np - (XL @ w)
                    pca_resid = X_train_np - (XL @ L.T)
                    lr_obj = np.mean(lr_resid ** 2)
                    pca_obj = np.mean(pca_resid ** 2)
                    return (lambda * lr_obj) + ((1 - lambda) * pca_obj)

                problem = Problem(manifold=manifold, cost=cost)
                solver = SteepestDescent(verbosity=0)
                L_init = onp.random.normal(size=(p, n_components))
                w_init = onp.random.normal(size=(n_components, 1))
                try:
                    result = solver.run(problem, initial_point=[L_init, w_init])
                    L_optimized, w_optimized = result.point
                    Z_val_man = X_val_np @ L_optimized
                    Y_pred_man = Z_val_man @ w_optimized
                    current_mse = mean_squared_error(Y_val, Y_pred_man)
                    
                    if current_mse < best_lambda_mse:
                        best_lambda_mse = current_mse
                        best_lambda = lambda
                except:
                    continue
            
        if best_lambda is not None:
                manifold = Product([Grassmann(p, n_components), Euclidean(n_components, 1)])
                @pymanopt.function.autograd(manifold)
                def cost(L, w):
                    XL = X_train_np @ L
                    lr_resid = Y_train_np - (XL @ w)
                    pca_resid = X_train_np - (XL @ L.T)
                    lr_obj = np.mean(lr_resid ** 2)
                    pca_obj = np.mean(pca_resid ** 2)
                    return (best_lambda * lr_obj) + ((1 - best_lambda) * pca_obj)

                problem = Problem(manifold=manifold, cost=cost)
                solver = SteepestDescent(verbosity=0)
                L_init = onp.random.normal(size=(p, n_components))
                w_init = onp.random.normal(size=(n_components, 1))
                try:
                    result = solver.run(problem, initial_point=[L_init, w_init])
                    L_optimized, w_optimized = result.point
                    Z_train_man = X_train @ L_optimized
                    Z_test_man = X_test @ L_optimized
                    Y_pred_man = Z_test_man @ w_optimized
                    expl_var_man = onp.trace(L_optimized.T @ X_train.T @ X_train @ L_optimized)
                    metrics['variance_explained_manifold'].append(expl_var_man / total_var)
                    metrics['mse_Manifold'].append(mean_squared_error(Y_test, Y_pred_man))
                    cov_zy = onp.cov(Z_train_man.T, Y_train.T)[:-1, -1]
                    metrics['covariance_explained_Manifold'].append(onp.sum(cov_zy**2) / onp.sum(cov_xy**2))
                    best_params['lambda']['value'].append(best_lambda)
                    best_params['lambda']['mse'].append(best_lambda_mse)
                except:
                    metrics['variance_explained_lspca'].append(0)
                    metrics['mse_LSPCA'].append(float('inf'))
                    metrics['covariance_explained_LSPCA'].append(0)
                    best_params['lambda']['value'].append(lambda_list[0])
                    best_params['lambda']['mse'].append(float('inf'))
        else:
                metrics['variance_explained_lspca'].append(0)
                metrics['mse_LSPCA'].append(float('inf'))
                metrics['covariance_explained_LSPCA'].append(0)
                best_params['lambda']['value'].append(lambda_list[0])
                best_params['lambda']['mse'].append(float('inf'))

        for key in metrics:
            results.loc[i, key] = onp.mean(metrics[key])
        
        results.loc[i, 'best_threshold'] = onp.mean(best_params['threshold']['value'])
        results.loc[i, 'best_kappa'] = onp.mean(best_params['kappa']['value'])
        results.loc[i, 'best_lambda'] = onp.mean(best_params['lambda']['value'])

    return results



# Simulation A: Linear (Y=3X_1-5X_2+4X_3+ε), i.i.d. scenario
def run_simulation_A(n_datasets=50, n=100, p=600, n_components_list=[2,4,6,8,10],
                    manifold_lambda_list=[],
                    lambda_list=[],
                    threshold_list=[],
                    seed=1234):
    methods = ['PCR', 'PLS', 'HSIC', 'Bair', 'CSPCA', 'LSPCA']
    results = {method: {
        'VarExpl': onp.zeros((n_datasets, len(n_components_list))),
        'MSE': onp.zeros((n_datasets, len(n_components_list))),
        'CovExpl': onp.zeros((n_datasets, len(n_components_list))),
        'BestParams': [None] * len(n_components_list)  
    } for method in methods}

    for i in range(n_datasets):
        onp.random.seed(seed + i)
        X = onp.random.normal(size=(n, p))
        X_informative = X[:, :3]
        true_beta = onp.array([3, -5, 4])
        error = onp.random.normal(0, 0.1, n)
        Y = X_informative @ true_beta + error
        data_A = onp.column_stack((Y[:, onp.newaxis], X))

        #For the correlated scenario we used
        #def create_toeplitz_cov(p, rho):
        #cov = onp.zeros((p, p))
        #for i in range(p):
        #    for j in range(p):
        #        cov[i, j] = rho ** abs(i - j)
        #return cov
        #cov_matrix = create_toeplitz_cov(p, rho)
        #X = onp.random.multivariate_normal(mean=onp.zeros(p), cov=cov_matrix, size=n)
        #X_informative = X[:, :3]
        #true_beta = onp.array([3, -5, 4])
        #error = onp.random.normal(0, 0.1, n)
        #Y = X_informative @ true_beta + error
        #data_A = onp.column_stack((Y[:, onp.newaxis], X))  


        res = run_all_methods_analysis(
            data_A, 
            n_components_list=n_components_list,
            lambda_list=lambda_list,
            kappa_list=kappa_list,
            threshold_list=threshold_list
        )

        for method in methods:
            var_key = f'variance_explained_{method.lower()}'
            results[method]['VarExpl'][i] = res[var_key]
            results[method]['MSE'][i] = res[f'mse_{method}']
            if method != 'PLS':
                results[method]['CovExpl'][i] = res[f'covariance_explained_{method}']
            
            if method == 'Bair':
                results[method]['BestParams'] = res['best_threshold'].tolist()
            elif method == 'CSPCA':
                results[method]['BestParams'] = res['best_kappa'].tolist()
            elif method == 'LSPCA':
                results[method]['BestParams'] = res['best_lambda'].tolist()

    
    def compute_stats(metric_matrix):
        avg = onp.nanmean(metric_matrix, axis=0)
        std_err = onp.nanstd(metric_matrix, axis=0) / onp.sqrt(n_datasets)
        return {'avg': avg, 'std_err': std_err}

    stats = {method: {
        'VarExpl': compute_stats(results[method]['VarExpl']),
        'MSE': compute_stats(results[method]['MSE']),
        'CovExpl': compute_stats(results[method]['CovExpl']),
        'BestParams': results[method]['BestParams']  
    } for method in methods}

    print("\nSimulation A (Linear: Y = 3X1 - 5X2 + 4X3 + error):")
    for method in methods:
        print(f"\n  {method}:")
        for metric in ['Variance Explained', 'MSE', 'Covariance Explained']:
            print(f"    {metric}:")
            for j, n_comp in enumerate(n_components_list):
                key = metric.split()[0][:3] + ('Expl' if 'Explained' in metric else '')
                avg = stats[method][key]['avg'][j]
                se = stats[method][key]['std_err'][j]
                print(f"      n_components = {n_comp}: Avg = {avg:.4f} ± {se:.4f}")
        
        if method in ['Bair', 'CSPCA', 'LSPCA']:
            print("    Best Parameters:")
            for j, n_comp in enumerate(n_components_list):
                param_name = {
                    'Bair': 'Threshold',
                    'CSPCA': 'Kappa',
                    'LSPCA': 'Lambda'
                }[method]
                param_value = stats[method]['BestParams'][j]
                print(f"      n_components = {n_comp}: {param_name} = {param_value:.4f}")

    return {'stats': stats, 'raw_results': results}
