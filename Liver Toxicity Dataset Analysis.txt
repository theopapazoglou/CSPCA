import numpy as onp
import autograd.numpy as np
import pandas as pd
from sklearn.decomposition import PCA
from sklearn.cross_decomposition import PLSRegression
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error, r2_score
from sklearn.preprocessing import StandardScaler
from scipy.spatial.distance import cdist
from scipy import linalg

from pymanopt.manifolds import Grassmann, Euclidean, Product
from pymanopt import Problem

import pymanopt.function
from scipy.linalg import svd, qr, pinv


def rbf_kernel(Y, sigma):
    dist_matrix = cdist(Y, Y, 'euclidean')
    K = np.exp(-dist_matrix**2 / (2 * sigma**2))
    return K

def barshan_pca(X, Y, K, q):
    Delta = linalg.cholesky(K, lower=True)  
    psi = X.T @ Delta  
    U, s, Vt = svd(psi, full_matrices=False)
    Sigma = onp.diag(s)
    U_d = U[:, :q]
    Sigma_d = Sigma[:q, :q]
    V_d = Vt.T[:, :q]  
    U_hat = psi @ V_d @ linalg.inv(Sigma_d)  
    eigenvalues = s[:q]**2  
    return {'W': U_hat, 'singular_values': eigenvalues}


def supervised_pca(X, Y, q, p, m, lambda_):
    XtY = X.T @ Y
    YtX = Y.T @ X
    XtX = X.T @ X
    C = XtY @ YtX + lambda_ * XtX  
    indices = onp.random.choice(p, size=m, replace=False)
    C_nm = C[:, indices]  
    C_mm = C[indices, :][:, indices]  
    eigenvalues, U_m = linalg.eigh(C_mm)  
    sorted_indices = onp.argsort(eigenvalues)[::-1]
    eigenvalues = eigenvalues[sorted_indices]
    U_m = U_m[:, sorted_indices]
    Lambda_m = onp.diag(eigenvalues)
    Lambda_m_inv_sqrt = onp.diag(1.0 / onp.sqrt(onp.maximum(eigenvalues, 1e-6)))
    U_n = C_nm @ U_m @ Lambda_m_inv_sqrt  
    C_mm_inv = pinv(C_mm)  
    C_approx = C_nm @ C_mm_inv @ C_nm.T
    W = U_n[:, :q]
    W, _ = qr(W, mode='economic')  
    return {'W': W, 'eigenvalues': eigenvalues[:q]}

def run_all_methods_analysis(data, n_splits=10, n_components_list=[2,4,6,8,10],
                             manifold_lambda=_, lambda_=_, threshold=_, seed=1888):
    
    Y = data[:, 0:1]  
    X = data[:, 1:]
    p = X.shape[1]

    results = {n_comp: {key: [] for key in [
        'total_variance_explained_pcr', 'mse_PCR', 'R2_PCR', 'covariance_explained_PCR',
        'variance_explained_pls', 'mse_PLS', 'R2_PLS',
        'variance_explained_barshan', 'mse_Barshan', 'R2_Barshan', 'covariance_explained_Barshan',
        'variance_explained_bair', 'mse_Bair', 'R2_Bair', 'covariance_explained_Bair',
        'variance_explained_supervised', 'mse_Supervised', 'R2_Supervised', 'covariance_explained_Supervised',
        'variance_explained_manifold', 'mse_Manifold', 'R2_Manifold', 'covariance_explained_Manifold'
    ]} for n_comp in n_components_list}

    for n_components in n_components_list:
        for i in range(n_splits):
            current_seed = seed + i if seed is not None else None
            X_train, X_test, Y_train, Y_test = train_test_split(
                X, Y, test_size=0.2, random_state=current_seed
            )

            scaler_X = StandardScaler()
            X_train = scaler_X.fit_transform(X_train)
            X_test = scaler_X.transform(X_test)
            scaler_Y = StandardScaler()
            Y_train = scaler_Y.fit_transform(Y_train)
            Y_test = scaler_Y.transform(Y_test)
            X_train_np = np.array(X_train)
            X_test_np = np.array(X_test)
            Y_train_np = np.array(Y_train)
            Y_test_np = np.array(Y_test)

            # PCR
            pca = PCA(n_components=n_components)
            X_train_pca = pca.fit_transform(X_train)
            X_test_pca = pca.transform(X_test)
            var_explained = onp.sum(pca.explained_variance_ratio_)
            results[n_components]['total_variance_explained_pcr'].append(var_explained)
            lr = LinearRegression()
            lr.fit(X_train_pca, Y_train)
            Y_pred = lr.predict(X_test_pca)
            results[n_components]['mse_PCR'].append(mean_squared_error(Y_test, Y_pred))
            results[n_components]['R2_PCR'].append(r2_score(Y_test, Y_pred))
            cov_xy = onp.cov(X_train.T, Y_train.T)[:-1, -1]
            cov_zy = onp.cov(X_train_pca.T, Y_train.T)[:-1, -1]
            results[n_components]['covariance_explained_PCR'].append(onp.sum(cov_zy**2) / onp.sum(cov_xy**2))

            # PLS
            pls = PLSRegression(n_components=n_components)
            pls.fit(X_train, Y_train)
            Y_pred_pls = pls.predict(X_test)
            var_expl_pls = onp.var(pls.x_scores_, axis=0).sum() / onp.var(X_train, axis=0).sum()
            results[n_components]['variance_explained_pls'].append(var_expl_pls)
            results[n_components]['mse_PLS'].append(mean_squared_error(Y_test, Y_pred_pls))
            results[n_components]['R2_PLS'].append(r2_score(Y_test, Y_pred_pls))

            # SPCA using HSIC
            sigma = 0.1
            K = rbf_kernel(Y_train, sigma)
            epsilon = 1e-15  
            K += epsilon * np.eye(K.shape[0])
            spca_result = barshan_pca(X_train, Y_train, K, n_components)
            W_barshan = spca_result['W']
            Z_train_bar = X_train @ W_barshan
            Z_test_bar = X_test @ W_barshan
            total_var = onp.trace(X_train.T @ X_train)
            expl_var = onp.trace(W_barshan.T @ X_train.T @ X_train @ W_barshan)
            results[n_components]['variance_explained_barshan'].append(expl_var / total_var)
            lr_barshan = LinearRegression().fit(Z_train_bar, Y_train)
            Y_pred_bar = lr_barshan.predict(Z_test_bar)
            results[n_components]['mse_Barshan'].append(mean_squared_error(Y_test, Y_pred_bar))
            results[n_components]['R2_Barshan'].append(r2_score(Y_test, Y_pred_bar))
            cov_zy = onp.cov(Z_train_bar.T, Y_train.T)[:-1, -1]
            results[n_components]['covariance_explained_Barshan'].append(onp.sum(cov_zy**2) / onp.sum(cov_xy**2))

            # Bair's Method
            feature_scores = onp.abs(onp.corrcoef(X_train.T, Y_train.T)[:-1, -1])
            selected_features = feature_scores >= threshold
            X_train_selected = X_train[:, selected_features]
            X_test_selected = X_test[:, selected_features]
            pca_bair = PCA(n_components=min(n_components, X_train_selected.shape[1]))
            X_train_bair = pca_bair.fit_transform(X_train_selected)
            X_test_bair = pca_bair.transform(X_test_selected)
            expl_var_bair = onp.trace(X_train_bair.T @ X_train_bair)
            results[n_components]['variance_explained_bair'].append(expl_var_bair / total_var)
            lr_bair = LinearRegression().fit(X_train_bair, Y_train)
            Y_pred_bair = lr_bair.predict(X_test_bair)
            results[n_components]['mse_Bair'].append(mean_squared_error(Y_test, Y_pred_bair))
            results[n_components]['R2_Bair'].append(r2_score(Y_test, Y_pred_bair))
            cov_zy = onp.cov(X_train_bair.T, Y_train.T)[:-1, -1]
            results[n_components]['covariance_explained_Bair'].append(onp.sum(cov_zy**2) / onp.sum(cov_xy**2))

            # CSPCA
            spca_sup = supervised_pca(X_train, Y_train, n_components, p=p, m=240, lambda_=lambda_)
            W_sup = spca_sup['W']
            Z_train_sup = X_train @ W_sup
            Z_test_sup = X_test @ W_sup
            expl_var_sup = onp.trace(W_sup.T @ X_train.T @ X_train @ W_sup)
            results[n_components]['variance_explained_supervised'].append(expl_var_sup / total_var)
            lr_sup = LinearRegression().fit(Z_train_sup, Y_train)
            Y_pred_sup = lr_sup.predict(Z_test_sup)
            results[n_components]['mse_Supervised'].append(mean_squared_error(Y_test, Y_pred_sup))
            results[n_components]['R2_Supervised'].append(r2_score(Y_test, Y_pred_sup))
            cov_zy = onp.cov(Z_train_sup.T, Y_train.T)[:-1, -1]
            results[n_components]['covariance_explained_Supervised'].append(onp.sum(cov_zy**2) / onp.sum(cov_xy**2))

            # LSPCA
            manifold = Product([Grassmann(p, n_components), Euclidean(n_components, 1)]) 
            @pymanopt.function.autograd(manifold)
            def cost(L, w):
                XL = X_train_np @ L
                lr_resid = Y_train_np - (XL @ w)
                pca_resid = X_train_np - (XL @ L.T)
                lr_obj = np.mean(lr_resid ** 2)
                pca_obj = np.mean(pca_resid ** 2)
                return (manifold_lambda * lr_obj) + ((1 - manifold_lambda) * pca_obj)

            problem = Problem(manifold=manifold, cost=cost)
            solver = SteepestDescent(verbosity=0)
            L_init = onp.random.normal(size=(p, n_components))
            w_init = onp.random.normal(size=(n_components, 1))
            result = solver.run(problem, initial_point=[L_init, w_init])
            L_optimized, w_optimized = result.point
            Z_train_man = X_train @ L_optimized
            Z_test_man = X_test @ L_optimized
            Y_pred_man = Z_test_man @ w_optimized
            expl_var_man = onp.trace(L_optimized.T @ X_train.T @ X_train @ L_optimized)
            results[n_components]['variance_explained_manifold'].append(expl_var_man / total_var)
            results[n_components]['mse_Manifold'].append(mean_squared_error(Y_test, Y_pred_man))
            results[n_components]['R2_Manifold'].append(r2_score(Y_test, Y_pred_man))
            cov_zy = onp.cov(Z_train_man.T, Y_train.T)[:-1, -1]
            results[n_components]['covariance_explained_Manifold'].append(onp.sum(cov_zy**2) / onp.sum(cov_xy**2))

    
    results_df = pd.DataFrame({
        'n_components': n_components_list,
        'total_variance_explained_pcr': [onp.mean(results[n_comp]['total_variance_explained_pcr']) for n_comp in n_components_list],
        'mse_PCR': [onp.mean(results[n_comp]['mse_PCR']) for n_comp in n_components_list],
        'R2_PCR': [onp.mean(results[n_comp]['R2_PCR']) for n_comp in n_components_list],
        'covariance_explained_PCR': [onp.mean(results[n_comp]['covariance_explained_PCR']) for n_comp in n_components_list],
        'variance_explained_pls': [onp.mean(results[n_comp]['variance_explained_pls']) for n_comp in n_components_list],
        'mse_PLS': [onp.mean(results[n_comp]['mse_PLS']) for n_comp in n_components_list],
        'R2_PLS': [onp.mean(results[n_comp]['R2_PLS']) for n_comp in n_components_list],
        'variance_explained_barshan': [onp.mean(results[n_comp]['variance_explained_barshan']) for n_comp in n_components_list],
        'mse_Barshan': [onp.mean(results[n_comp]['mse_Barshan']) for n_comp in n_components_list],
        'R2_Barshan': [onp.mean(results[n_comp]['R2_Barshan']) for n_comp in n_components_list],
        'covariance_explained_Barshan': [onp.mean(results[n_comp]['covariance_explained_Barshan']) for n_comp in n_components_list],
        'variance_explained_bair': [onp.mean(results[n_comp]['variance_explained_bair']) for n_comp in n_components_list],
        'mse_Bair': [onp.mean(results[n_comp]['mse_Bair']) for n_comp in n_components_list],
        'R2_Bair': [onp.mean(results[n_comp]['R2_Bair']) for n_comp in n_components_list],
        'covariance_explained_Bair': [onp.mean(results[n_comp]['covariance_explained_Bair']) for n_comp in n_components_list],
        'variance_explained_supervised': [onp.mean(results[n_comp]['variance_explained_supervised']) for n_comp in n_components_list],
        'mse_Supervised': [onp.mean(results[n_comp]['mse_Supervised']) for n_comp in n_components_list],
        'R2_Supervised': [onp.mean(results[n_comp]['R2_Supervised']) for n_comp in n_components_list],
        'covariance_explained_Supervised': [onp.mean(results[n_comp]['covariance_explained_Supervised']) for n_comp in n_components_list],
        'variance_explained_manifold': [onp.mean(results[n_comp]['variance_explained_manifold']) for n_comp in n_components_list],
        'mse_Manifold': [onp.mean(results[n_comp]['mse_Manifold']) for n_comp in n_components_list],
        'R2_Manifold': [onp.mean(results[n_comp]['R2_Manifold']) for n_comp in n_components_list],
        'covariance_explained_Manifold': [onp.mean(results[n_comp]['covariance_explained_Manifold']) for n_comp in n_components_list]
    })

    return {'per_split_results': results, 'averaged_results': results_df}

def run_real_data_analysis(data, n_splits=20, n_components_list=[2,4,6,8,10]):
    methods = ['PCR', 'PLS', 'Barshan', 'Bair', 'Supervised', 'Manifold']
    results = {method: {
        'VarExpl': onp.zeros((len(n_components_list), n_splits)),
        'MSE': onp.zeros((len(n_components_list), n_splits)),
        'CovExpl': onp.zeros((len(n_components_list), n_splits)),
        'R2': onp.zeros((len(n_components_list), n_splits))
    } for method in methods}

    analysis_results = run_all_methods_analysis(data, n_splits=n_splits, n_components_list=n_components_list)
    per_split_results = analysis_results['per_split_results']
    for j, n_comp in enumerate(n_components_list):
        for method in methods:
            var_key = 'total_variance_explained_pcr' if method == 'PCR' else f'variance_explained_{method.lower()}'
            mse_key = f'mse_{method}'
            r2_key = f'R2_{method}'
            cov_key = f'covariance_explained_{method}'

            results[method]['VarExpl'][j, :] = per_split_results[n_comp][var_key]
            results[method]['MSE'][j, :] = per_split_results[n_comp][mse_key]
            results[method]['R2'][j, :] = per_split_results[n_comp][r2_key]
            if method != 'PLS':  # PLS doesn't compute covariance explained
                results[method]['CovExpl'][j, :] = per_split_results[n_comp][cov_key]

    def compute_stats(metric_matrix):
        avg = onp.nanmean(metric_matrix, axis=1)
        std_err = onp.nanstd(metric_matrix, axis=1) / onp.sqrt(n_splits)
        return {'avg': avg, 'std_err': std_err}

    stats = {method: {
        'VarExpl': compute_stats(results[method]['VarExpl']),
        'MSE': compute_stats(results[method]['MSE']),
        'CovExpl': compute_stats(results[method]['CovExpl']),
        'R2': compute_stats(results[method]['R2'])
    } for method in methods}

    print("\nLiver Toxicity Data Analysis (20 Train-Test Splits):")
    for method in methods:
        print(f"  {method}:")
        for metric in ['Variance Explained', 'MSE', 'Covariance Explained', 'R2']:
            if metric == 'Covariance Explained' and method == 'PLS':
                continue  
            print(f"    {metric}:")
            for j, n_comp in enumerate(n_components_list):
                key = metric.split()[0][:3] + ('Expl' if 'Explained' in metric else '')
                avg = stats[method][key]['avg'][j]
                se = stats[method][key]['std_err'][j]
                print(f"      n_components = {n_comp}: Avg = {avg:.4f} ± {se:.4f}")

    return {'stats': stats, 'raw_results': results, 'averaged_results': analysis_results['averaged_results']}


Y_df = pd.read_csv('liver_toxicity_Y.csv')
X_df = pd.read_csv('liver_toxicity_X.csv')
Y = Y_df.values
X = X_df.values
if len(Y.shape) == 1 or Y.shape[1] == 1:
    Y = Y.reshape(-1, 1)
print("Y shape:", Y.shape)
print("X shape:", X.shape)
data = onp.column_stack((Y, X))
onp.random.seed(12561)
results = run_real_data_analysis(data, n_splits=20, n_components_list=[2,4,6,8,10])
